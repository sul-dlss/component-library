Popovers

Bootstrap 6 View the Bootstrap 5 version

This is a dimissible popover that doesn't disappear when users copy / interact with the popover

The button/link that triggers the popover must have a unique "id" attribute

HTML
You must initialize this with this javascript module (Bootstrap 6 ships ES modules only, with no bootstrap global): 
import { Popover } from 'https://cdn.jsdelivr.net/npm/bootstrap@6.0.0-alpha.1/dist/js/bootstrap.bundle.min.js'

document.querySelectorAll('[data-bs-toggle="popover"]').forEach((popoverTriggerEl) => {
  new Popover(popoverTriggerEl, {
    template: `<div class="popover" role="tooltip" data-controller="popover">
                <div class="popover-arrow"></div>
                <div class="popover-header">
                </div>
                <div class="popover-body">
                </div>
                </div>`,
    title: (btn) => `<div class="d-flex justify-content-between align-content-center">
    <div class="fw-semibold align-self-center">${btn.dataset.bsTitle}:</div>
    <button class="btn-text p-0 ms-3 fs-lg lh-1" aria-label="Close this thing" data-popover-dismiss="${btn.id}" type="button">
      <i class="bi bi-x"></i>
    </button></div>`,
    html: true,
    // tags and attributes allowed in the popover, everything not listed will be stripped
    // needed to be able to put a data-controller and data-action in popover
    allowList: {
      '*': ['class', 'role', /^aria-[\w-]*$/i],
      div: ['class', 'role'],
      button: ['data-popover-dismiss', 'type'],
      i: ['class']
    }
  })
})

// There is no global `bootstrap` object in Bootstrap 6, so the close button can't call
// bootstrap.Popover in an onclick attribute. Listen for it here instead.
document.addEventListener('click', (event) => {
  const closeButton = event.target.closest('[data-popover-dismiss]')
  if (closeButton) Popover.getInstance(document.getElementById(closeButton.dataset.popoverDismiss))?.hide()
})

Notes

  • the arrow can be left, right, top or bottom by setting data-bs-placement. See https://getbootstrap.com/docs/6.0/components/popover/#placement
